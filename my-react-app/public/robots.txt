# The public content surfaces. Both are explicitly allowed because the
# Disallow list below is prefix-matched and /questions sits uncomfortably close
# to /question-reports; spelling it out means a future edit to that line cannot
# silently de-index 250 pages.
Allow: /guides
Allow: /questions

# Admin surfaces — never useful to a crawler, and cheaper to exclude here than
# to rely on the noindex meta tag alone.
Disallow: /admin
Disallow: /ADD_ACCOUNT
Disallow: /ADDQ
Disallow: /Bank
Disallow: /TEMP_LINKS
Disallow: /question-reports

# Signed-in app pages. A crawler only ever sees the login redirect, so these
# burn crawl budget and look like thin duplicate pages.
Disallow: /quizs
Disallow: /quiz/
Disallow: /analysis
Disallow: /wrong-questions
Disallow: /summaries
Disallow: /subscribe
Disallow: /payment/

# Auth utility pages with nothing to rank for.
#
# /signup and /login are deliberately NOT disallowed here even though both
# serve `noindex`, and that combination is intentional rather than a
# contradiction: a URL that robots.txt blocks cannot be fetched, so Googlebot
# never reads the noindex on it and may index the bare URL anyway. Allowing the
# crawl is what makes the noindex effective. Keep them out of the sitemap
# (postbuild-seo.mjs already filters noindex routes) and out of this list.
#
# Only the per-invite token URLs under /signup/ are excluded, since each is a
# single-use link with nothing to crawl.
Disallow: /forgot-password
Disallow: /signup/
